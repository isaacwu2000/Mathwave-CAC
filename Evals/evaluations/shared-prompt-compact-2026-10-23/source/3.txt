export type ToolName = keyof typeof definitions;
export const toolCatalog: Readonly<Record<ToolName, ToolDefinition>> =
  definitions;
export type ToolScope =
  | 'interpreter'
  | 'graph'
  | 'graph-scene'
  | 'construction';

/** Resolves only registered calls, including the historical notebook alias; unknown names fail closed. */
export function programTool(requestedName: string) {
  const name = requestedName.replace(/^notebook\./, 'problem.');
  if (!Object.hasOwn(toolCatalog, name))
    throw new Error(`Unknown function ${name}.`);
  return {
    name: name as ToolName,
    definition: toolCatalog[name as ToolName] as ToolDefinition,
  };
}

/** Uses execution categories to restrict graph builds; saved results receive an additional state check. */
export function isVisualTool(name: string) {
  const { definition: tool } = programTool(name);
  return (
    tool.kind !== 'control' &&
    (tool.kind === 'save' || tool.area === 'graph' || tool.area === 'geometry')
  );
}

const parameterTypes = {
  number: 'number',
  string: 'string',
  numbers: 'number[]',
  strings: 'string[]',
  dimensions: 'numericObject',
  operand: 'number|string|MathValue',
  math: 'MathValue',
  part: 'PartReference',
  mathOrPart: 'MathValue|PartReference',
  graph: 'GraphValue',
  geometry: 'ShapeValue',
  statistics: 'DataValue',
  diagram: 'DiagramValue',
  work: 'WorkValue',
  graphSelector: 'number|string',
} as const;

/** Renders signatures from executable contracts, grouping identical descriptions without duplicating APIs. */
export function toolInstructions(scope: ToolScope = 'interpreter') {
  const groups = new Map<string, { names: string[]; tool: ToolDefinition }>();
  for (const [name, definition] of Object.entries(toolCatalog)) {
    const tool: ToolDefinition = definition;
    if (
      scope === 'graph'
        ? !isVisualTool(name)
        : scope === 'interpreter'
          ? !!tool.context
          : tool.context !== scope
    )
      continue;
    const key = JSON.stringify(tool);
    const group = groups.get(key);
    if (group) group.names.push(name);
    else groups.set(key, { names: [name], tool });
  }
  return [...groups.values()]
    .map(({ names, tool }) => {
      const parameters = tool.parameters
        .map(
          (p) => `${p.name}${p.optional ? '?' : ''}:${parameterTypes[p.type]}`,
        )
        .join(',');
      return `${names.join('|')}(${parameters}) -> ${tool.returns}. ${tool.description}${tool.constraints ? ' ' + tool.constraints : ''}${tool.examples ? ' Example: ' + tool.examples.join(' ') : ''}`;
    })
    .join('\n');
}

/** Gives Live concise summaries of registered backend capabilities rather than low-level call syntax. */
export function voiceCapabilities() {
  const areas = new Set<ToolArea>(
    Object.values(toolCatalog).map((tool) => tool.area),
  );
  return [...areas].map((area) => capabilityDescriptions[area]).join('\n');
}
