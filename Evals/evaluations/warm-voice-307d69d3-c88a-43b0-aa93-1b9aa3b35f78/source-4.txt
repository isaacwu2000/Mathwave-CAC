import { journeySpan, journeyServerTiming } from '../../diagnostics/index.ts';
import { rms, type AudioFrame } from '../services/silence-monitor.ts';
import { microphoneFailure } from './microphone.ts';
import type { LiveContext } from '../types/index.ts';
export type LocalCapture = {
  stream: MediaStream;
  play(frame: AudioFrame): void;
  draining(): boolean;
  close(): void;
};
export type LiveTransport = {
  send(event: object): void;
  dispose(): void;
  speaking(): boolean;
  replaceCapture?(capture: LocalCapture): Promise<void>;
  mute(value: boolean): void;
};
export type VoiceEnvironment = {
  configured(): Promise<boolean>;
  monitor(
    sample: (frame: AudioFrame) => void,
    signal?: AbortSignal,
  ): Promise<LocalCapture>;
  connect(
    capture: LocalCapture,
    context: LiveContext,
    event: (event: Record<string, unknown>) => void,
    journeyId?: string,
  ): Promise<LiveTransport>;
};

export const browserVoiceEnvironment: VoiceEnvironment = {
  async configured() {
    const response = await fetch('/api/live-token', { cache: 'no-store' });
    const body = (await response.json()) as { configured?: boolean };
    return response.ok && body.configured === true;
  },
  async monitor(sample, signal) {
    signal?.throwIfAborted();
    const context = new AudioContext();
    let stream: MediaStream | undefined;
    let closed = false;
    const closeInput = () => {
      if (closed) return;
      closed = true;
      stream?.getTracks().forEach((track) => track.stop());
      void context.close().catch(() => {});
    };
    let cancel!: () => void;
    const cancelled = new Promise<never>((_, reject) => {
      cancel = () => {
        closeInput();
        reject(
          signal?.reason ??
            new DOMException('Capture cancelled.', 'AbortError'),
        );
      };
    });
    signal?.addEventListener('abort', cancel, { once: true });
    try {
      // Start audio during the Record click; waiting for permission can lose browser activation.
      signal?.throwIfAborted();
      const resumed = context.resume();
      const microphone = navigator.mediaDevices
        .getUserMedia({
          audio: { echoCancellation: true, noiseSuppression: true },
        })
        .catch((error: unknown) => {
          throw microphoneFailure(error);
        })
        .then((granted) => {
          if (closed) {
            granted.getTracks().forEach((track) => track.stop());
            throw (
              signal?.reason ??
              new DOMException('Capture cancelled.', 'AbortError')
            );
          }
          stream = granted;
          return granted;
        });
      await Promise.race([Promise.all([resumed, microphone]), cancelled]);
      await Promise.race([
        context.audioWorklet.addModule('/voice-monitor.js'),
        cancelled,
      ]);
      signal?.throwIfAborted();
      const input = context.createMediaStreamSource(stream!);
      const worklet = new AudioWorkletNode(context, 'voice-monitor');
      const silent = context.createGain();
      silent.gain.value = 0;
      input.connect(worklet);
      worklet.connect(silent);
      silent.connect(context.destination);
      const outgoing = context.createMediaStreamDestination();
      let scheduledUntil = 0;
      const sources = new Set<AudioBufferSourceNode>();
      worklet.port.onmessage = (message) =>
        sample({
          samples: message.data as Float32Array,
          sampleRate: context.sampleRate,
          at: performance.now(),
        });
      return {
        stream: outgoing.stream,
        play(frame) {
          const buffer = context.createBuffer(
            1,
            frame.samples.length,
            frame.sampleRate,
          );
          buffer.copyToChannel(new Float32Array(frame.samples), 0);
          const source = context.createBufferSource();
          source.buffer = buffer;
          source.connect(outgoing);
          scheduledUntil = Math.max(context.currentTime + 0.02, scheduledUntil);
          source.start(scheduledUntil);
          scheduledUntil += buffer.duration;
          sources.add(source);
          source.onended = () => {
            sources.delete(source);
            source.disconnect();
          };
        },
        draining() {
          return scheduledUntil > context.currentTime + 0.04;
        },
        close() {
          worklet.port.onmessage = null;
          worklet.disconnect();
          input.disconnect();
          silent.disconnect();
          for (const source of sources) {
            source.stop();
            source.disconnect();
          }
          sources.clear();
          outgoing.stream.getTracks().forEach((track) => track.stop());
          closeInput();
        },
      };
    } catch (error) {
      closeInput();
      throw error;
    } finally {
      signal?.removeEventListener('abort', cancel);
    }
  },
  async connect(capture, context, receive, journeyId = crypto.randomUUID()) {
    const peer = new RTCPeerConnection();
    const channel = peer.createDataChannel('oai-events');
    const queuedEvents: string[] = [];
    channel.onopen = () => {
      for (const value of queuedEvents.splice(0)) channel.send(value);
    };
    const audio = new Audio();
    audio.autoplay = true;
    const playbackContext = new AudioContext();
    let speakerLastAt = -Infinity;
    let timer: ReturnType<typeof setInterval> | undefined;
    const dispose = () => {
      clearInterval(timer);
      audio.pause();
      audio.srcObject = null;
      channel.close();
      peer.close();
      void playbackContext.close().catch(() => {});
    };
    try {
      for (const track of capture.stream.getTracks())
        peer.addTrack(track, capture.stream);
      peer.ontrack = (event) => {
        const stream = event.streams[0] ?? new MediaStream([event.track]);
        audio.srcObject = stream;
        void audio.play().catch(() => receive({ type: 'playback.error' }));
        const source = playbackContext.createMediaStreamSource(stream);
        const analyser = playbackContext.createAnalyser();
        analyser.fftSize = 512;
        source.connect(analyser);
        const samples = new Float32Array(analyser.fftSize);
        let active = false;
        timer = setInterval(() => {
          analyser.getFloatTimeDomainData(samples);
          if (rms(samples) > 0.001) speakerLastAt = performance.now();
          const next = performance.now() - speakerLastAt < 350;
          if (next !== active) {
            active = next;
            receive({ type: 'playback.activity', active });
          }
        }, 25);
      };
      channel.onmessage = (message) => {
        try {
          receive(JSON.parse(message.data));
        } catch {
          receive({
            type: 'error',
            error: { message: 'Invalid voice response.' },
          });
        }
      };
      peer.onconnectionstatechange = () => {
        if (['failed', 'disconnected'].includes(peer.connectionState))
          receive({ type: 'transport.error' });
      };
      await playbackContext.resume();
      const offerStart = performance.now();
      await peer.setLocalDescription(await peer.createOffer());
      journeySpan(journeyId, 'webrtc-offer', offerStart);
      const iceStart = performance.now();
      if (peer.iceGatheringState !== 'complete')
        await new Promise<void>((resolve, reject) => {
          const timeout = setTimeout(() => {
            peer.removeEventListener('icegatheringstatechange', changed);
            reject(new Error('Voice connection timed out.'));
          }, 10000);
          const changed = () => {
            if (peer.iceGatheringState === 'complete') {
              clearTimeout(timeout);
              peer.removeEventListener('icegatheringstatechange', changed);
              resolve();
            }
          };
          peer.addEventListener('icegatheringstatechange', changed);
          changed();
        });
      journeySpan(journeyId, 'ice-gather', iceStart);
      const requestStart = performance.now();
      const response = await fetch('/api/live-token', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          ...(context.owner ? { 'X-Mathwave-Owner': context.owner } : {}),
        },
        body: JSON.stringify({
          sdp: peer.localDescription?.sdp,
          revision: context.revision,
        }),
        signal: AbortSignal.timeout(20000),
      });
      const body = (await response.json()) as { sdp?: string; error?: string };
      journeySpan(journeyId, 'live-token-request', requestStart);
      journeyServerTiming(
        journeyId,
        response.headers.get('Server-Timing'),
        'live-server',
      );
      if (!response.ok || !body.sdp)
        throw new Error(body.error || 'GPT Live could not connect.');
      await peer.setRemoteDescription({ type: 'answer', sdp: body.sdp });
      return {
        send(value) {
          const serialized = JSON.stringify(value);
          if (channel.readyState === 'open') channel.send(serialized);
          else if (channel.readyState === 'connecting')
            queuedEvents.push(serialized);
        },
        dispose,
        async replaceCapture(capture) {
          const track = capture.stream.getAudioTracks()[0];
          const sender = peer
            .getSenders()
            .find((item) => item.track?.kind === 'audio');
          if (!track || !sender)
            throw new Error('Microphone input could not reconnect.');
          await sender.replaceTrack(track);
        },
        speaking: () => performance.now() - speakerLastAt < 350,
        mute(value) {
          audio.muted = value;
        },
      };
    } catch (error) {
      dispose();
      throw error;
    }
  },
};
